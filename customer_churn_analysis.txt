DROP DATABASE IF EXISTS customer_churn_db;
CREATE DATABASE customer_churn_db;
USE customer_churn_db;

CREATE TABLE customers (
    customer_id INT PRIMARY KEY,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    age TINYINT UNSIGNED,
    gender VARCHAR(20),
    city VARCHAR(80),
    signup_date DATE NOT NULL,
    CONSTRAINT chk_customer_age CHECK (age IS NULL OR age BETWEEN 18 AND 110)
);

CREATE TABLE subscriptions (
    subscription_id INT PRIMARY KEY,
    customer_id INT NOT NULL,
    plan_name VARCHAR(30) NOT NULL,
    contract_type VARCHAR(30) NOT NULL,
    monthly_charge DECIMAL(10,2) NOT NULL,
    start_date DATE NOT NULL,
    end_date DATE NULL,
    status ENUM('Active','Churned','Paused') NOT NULL,
    CONSTRAINT fk_subscription_customer FOREIGN KEY (customer_id)
        REFERENCES customers(customer_id),
    CONSTRAINT chk_monthly_charge CHECK (monthly_charge >= 0),
    CONSTRAINT chk_subscription_dates CHECK (end_date IS NULL OR end_date >= start_date)
);

CREATE TABLE payments (
    payment_id INT PRIMARY KEY,
    customer_id INT NOT NULL,
    payment_date DATE NOT NULL,
    amount DECIMAL(10,2) NOT NULL,
    payment_method VARCHAR(30) NOT NULL,
    payment_status ENUM('Successful','Failed','Refunded') NOT NULL,
    CONSTRAINT fk_payment_customer FOREIGN KEY (customer_id)
        REFERENCES customers(customer_id),
    CONSTRAINT chk_payment_amount CHECK (amount >= 0)
);

CREATE TABLE support_tickets (
    ticket_id INT PRIMARY KEY,
    customer_id INT NOT NULL,
    opened_date DATE NOT NULL,
    issue_type VARCHAR(50) NOT NULL,
    priority ENUM('Low','Medium','High','Critical') NOT NULL,
    resolution_status ENUM('Open','In Progress','Resolved') NOT NULL,
    resolution_hours DECIMAL(8,2) NULL,
    CONSTRAINT fk_ticket_customer FOREIGN KEY (customer_id)
        REFERENCES customers(customer_id),
    CONSTRAINT chk_resolution_hours CHECK (resolution_hours IS NULL OR resolution_hours >= 0)
);

CREATE TABLE churn_events (
    churn_event_id INT PRIMARY KEY,
    customer_id INT NOT NULL,
    churn_date DATE NOT NULL,
    churn_reason VARCHAR(80) NOT NULL,
    feedback VARCHAR(255),
    CONSTRAINT fk_churn_customer FOREIGN KEY (customer_id)
        REFERENCES customers(customer_id),
    CONSTRAINT uq_customer_churn UNIQUE (customer_id)
);

-- View 1: one reusable customer-level churn summary.
CREATE OR REPLACE VIEW vw_customer_churn_summary AS
SELECT
    c.customer_id,
    CONCAT(c.first_name, ' ', c.last_name) AS customer_name,
    c.city,
    c.signup_date,
    s.plan_name,
    s.contract_type,
    s.monthly_charge,
    s.status AS subscription_status,
    ce.churn_date,
    ce.churn_reason,
    CASE WHEN ce.customer_id IS NULL THEN 0 ELSE 1 END AS churn_flag
FROM customers c
JOIN subscriptions s ON s.customer_id = c.customer_id
LEFT JOIN churn_events ce ON ce.customer_id = c.customer_id;

-- View 2: aggregate support and payment indicators per customer.
CREATE OR REPLACE VIEW vw_customer_service_summary AS
SELECT
    c.customer_id,
    CONCAT(c.first_name, ' ', c.last_name) AS customer_name,
    COALESCE(st.ticket_count, 0) AS ticket_count,
    COALESCE(st.high_priority_tickets, 0) AS high_priority_tickets,
    COALESCE(p.failed_payments, 0) AS failed_payments
FROM customers c
LEFT JOIN (
    SELECT customer_id,
           COUNT(*) AS ticket_count,
           SUM(priority IN ('High', 'Critical')) AS high_priority_tickets
    FROM support_tickets
    GROUP BY customer_id
) st ON st.customer_id = c.customer_id
LEFT JOIN (
    SELECT customer_id,
           SUM(payment_status = 'Failed') AS failed_payments
    FROM payments
    GROUP BY customer_id
) p ON p.customer_id = c.customer_id;

DROP PROCEDURE IF EXISTS sp_churn_report;
DELIMITER $$
CREATE PROCEDURE sp_churn_report(IN p_start_date DATE, IN p_end_date DATE)
BEGIN
    IF p_start_date IS NULL OR p_end_date IS NULL OR p_start_date > p_end_date THEN
        SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT = 'Provide a valid date range: start date must be <= end date';
    END IF;

    SELECT
        COUNT(*) AS churn_events_in_period,
        COUNT(DISTINCT customer_id) AS unique_customers_churned,
        MIN(churn_date) AS first_churn_date,
        MAX(churn_date) AS last_churn_date
    FROM churn_events
    WHERE churn_date BETWEEN p_start_date AND p_end_date;
END $$
DELIMITER ;

-- Audit table records changes to subscription status.
CREATE TABLE subscription_status_audit (
    audit_id INT AUTO_INCREMENT PRIMARY KEY,
    subscription_id INT NOT NULL,
    customer_id INT NOT NULL,
    old_status VARCHAR(20) NOT NULL,
    new_status VARCHAR(20) NOT NULL,
    changed_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- Trigger 1: when a churn event is inserted, synchronize the subscription.
DROP TRIGGER IF EXISTS trg_sync_subscription_after_churn;
DELIMITER $$
CREATE TRIGGER trg_sync_subscription_after_churn
AFTER INSERT ON churn_events
FOR EACH ROW
BEGIN
    UPDATE subscriptions
    SET status = 'Churned',
        end_date = NEW.churn_date
    WHERE customer_id = NEW.customer_id;
END $$
DELIMITER ;

-- Trigger 2: audit subscription status changes.
DROP TRIGGER IF EXISTS trg_audit_subscription_status;
DELIMITER $$
CREATE TRIGGER trg_audit_subscription_status
AFTER UPDATE ON subscriptions
FOR EACH ROW
BEGIN
    IF OLD.status <> NEW.status THEN
        INSERT INTO subscription_status_audit
            (subscription_id, customer_id, old_status, new_status)
        VALUES
            (NEW.subscription_id, NEW.customer_id, OLD.status, NEW.status);
    END IF;
END $$
DELIMITER ;

-- Confirm subscription status and churn-event records agree.
-- Expected result: empty set for the original sample data.
EXPLAIN
SELECT s.customer_id, s.status, ce.churn_date
FROM subscriptions s
LEFT JOIN churn_events ce ON ce.customer_id = s.customer_id
WHERE (s.status = 'Churned' AND ce.customer_id IS NULL)
   OR (s.status <> 'Churned' AND ce.customer_id IS NOT NULL);
